\begin{frame}{Coal mining in the US}
    \begin{itemize}
        \item ``Clean" coal, in fact, is not clean.
        \begin{itemize}
            \item Coal burning emits greenhouse gases and air pollution.
            \item Coal mining contaminates water.
        \end{itemize}
        \item The 1990 Clean Air Act Amendments, a.k.a. the Acid Rain Program (ARP), caused US coal demand to shift to low sulfur coal.
        \begin{itemize}
            \item SO2 and NOx emission limits on coal fired power plants (main consumers of US coal).
            \item Compliance deadline for most power plants by end of 1995 (stage 1).
            \item Compliance achieved with low sulfur coal.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Drinking water regulation}
    \begin{itemize}
        \item US coal mining's water pollution impacts depend on the responses of community water systems (CWS).
        \item CWSs regulated by the Safe Drinking Water Act (SDWA).
        \item CWSs self monitor and report water quality to regulators.
        \item CWSs violate the SDWA when they fail:
        \begin{itemize}
            \item To monitor their water and/or report tests to regulators on time (MR violation).
            \item Tested contaminant levels exceed the maximum contaminant level (MCL violation).
            \item MCL violations have greater penalties than MR violations.
        \end{itemize}
        \item Enforcement attempts to be remedial rather than punitive.
        \item CWS incentives to meet SDWA depend on the cost of compliance and the penalty of non-compliance.
    \end{itemize}
\end{frame}

\begin{frame}{Research questions}  
    \begin{itemize}
        \item How did the Acid Rain Program shift the burden of coal mining water pollution?
        \item Do CWSs stay in SDWA compliance with upstream coal mining pollution?
        \item Does coal mining raise water pollution at CWSs?
        \item Does coal mining raise enforcement at CWSs affected by coal mining?
    \end{itemize}
\end{frame}

\begin{frame}{What this paper does}
\begin{itemize}
    \item CWS panel dataset of
    \begin{itemize}
        \item SDWA violations
        \item Contaminant concentrations
        \item Water intakes
        \item Upstream coal production and characteristics
    \end{itemize}
    \item Identify populations relying on a CWS downstream of coal mining before and after ARP.

    \item 2SLS: coal mining on CWS SDWA violations
        \begin{itemize}
            \item Upstream coal mine instrument: upstream coal sulfur \% $\times$ post 1995 (ARP stage 1 deadline)
        \end{itemize}
        
    \item DiD: coal mining effect on CWS water pollution

    \item 2SLS: regulator actions towards CWSs affected by coal mining.

\end{itemize}    
\end{frame}

\begin{frame}{Results}
\begin{itemize}
\item Low sulfur coal areas had fewer CWS intakes, pre-vs-post ARP, 82\% fewer people receiving water from a CWS downstream of a coal mine.

\item CWS SDWA inorganic chemical MR violations (non-compliance) are higher due to coal mining.

\item $\uparrow$ coal mining $\rightarrow$ $\uparrow$ mean concentrations of inorganic chemicals (costs of compliance)
  
\item $\uparrow$ coal mining $\rightarrow$ $\downarrow$ formal enforcement (penalty of non-compliance) + $\uparrow$ sanitary visits (remedial enforcement)

\item CWS non-compliance increases as costs of compliance increase and penalty for non-compliance decreases
\end{itemize}
\end{frame}

\begin{frame}{IV Model: mining on CWS violations}
First stage:
\begin{equation}\label{1ststage2sls}
\begin{split}
\text{NumMines}_{ct} &= \alpha_0 + \alpha_1\text{NumIntakes}_{ct} + \alpha_2\text{UpstreamCoalSulfur\%}_c \cdot \text{Post95}_t \\
&\quad + Y_t + C_c + \nu_{ct}
\end{split}
\end{equation}

Second stage:
\begin{equation}\label{2ndstage2sls}
\begin{split}
\text{Days in Violation}_{vct} &= \beta_0 + \beta_1\text{NumIntakes}_{ct} + \gamma\widehat{\text{NumMines}}_{ct} \\
&\quad + Y_t + C_c + \epsilon_{ct}
\end{split}
\end{equation}

\begin{itemize}
    \begin{itemize}
        \item $Post95_t=$ 1 if year $>$ 1995 
        \item Standard errors are clustered at the CWS level
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{IV identification assumptions}
    \begin{itemize}
        \item Exclusion: $\text{UpstreamCoalSulfur\%}_c \cdot \text{Post95}_t$ did not affect violations outside of number of upstream mines.
        \item Independence: $\text{UpstreamCoalSulfur\%}_c \cdot \text{Post95}_t$, as good as random conditional on covariates. $\text{Post95}_t$ determined by the national CAAA legislation, and control for time invariant $\text{UpstreamCoalSulfur\%}$ with CWS fixed effects.
        \item Monotonicity/no defiers: A defier is a HUC12 that experienced more mining after 1995 because it had higher sulfur.
    \end{itemize}
\end{frame}

\begin{frame}{DiD Model: mining on CWS contaminant concentrations}
\begin{itemize}
    \item Panel data
    \begin{equation}\label{eq:huc02reg}
    \textit{Concen}_{cht}= \beta\cdot\sum_{\tau=1985}^t\text{UpstreamCoalProd}_{c\tau}  + \rho_c + HUC02_h\cdot\tau_t + \epsilon_{cht}
\end{equation}

    \begin{itemize}
        \item $\textit{Concen}_{cht}=$ mean contaminant concentration at CWS c in HUC02 h in year t.
        \item $\sum_{\tau=1985}^t\text{upstream coal prod}_{c\tau}=$ cumulative tons of coal mined upstream of CWS c in the HUC12 h from 1985 to year t.
        \item $\rho_c=$CWS fixed effects
        \item $HUC02_h\cdot\tau_t=$HUC02 interacted with time fixed effects
        \item Standard errors are clustered at the CWS level
    \end{itemize}
    \item Continuous DiD with no pure control observations. 
    \item Identification: CWSs that experience a large decline in coal production would have trended the same as CWSs with a small decline.
\end{itemize}
\end{frame}

\begin{frame}{Results}
\begin{itemize}
    \item Mining increases mean inorganic chemical contamination (by 40-80\% of mean concentrations).
    \item Mining raises CWS costs to completely remove contaminants.
    \item CWSs downstream of mines face costlier SDWA compliance. 
\end{itemize}
\end{frame}

\begin{frame}{Conclusion}
\begin{itemize}
    \item Mining induces lower CWS SDWA compliance.
    \item $\uparrow$ mining $\rightarrow$   $\uparrow$ inorganic chemical water contamination.
    \item $\uparrow$ inorganic chemical water contamination $\rightarrow$ $\uparrow$ CWS cost of SDWA compliance.
    \item $\uparrow$ mining $\rightarrow$ $\downarrow$ formal enforcement action (62\% decrease relative to mean formal enforcement rate) + $\uparrow$ sanitary visits (non-punitive enforcement).
    \item (Lower punishment for violations + higher costs of compliance) $\rightarrow$ reduced CWS SDWA compliance.
\end{itemize}
\end{frame}
