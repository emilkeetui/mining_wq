\begin{frame}{Self-reported compliance}
\begin{itemize}
    \item Self-reported regulation monitoring requires entities to demonstrate compliance to regulators.
    \item Incentives to shirk reporting:
    \begin{itemize}
        \item Concealing non-compliance
        \item Saving self-reporting costs
    \end{itemize}
    \item Environmental contamination regulation (e.g. US drinking water): 
    \begin{itemize}
        \item $\uparrow$ contamination $\rightarrow$ $\uparrow$ shirking incentives
    \end{itemize}
    
    \item \textbf{Why care about self-reporting and contamination?}
    \begin{itemize}
        \item Different fix if concealing non-compliance vs self-reporting costs
        \item Monitoring used across industries (HIPAA violation disclosure, SOX, energy firms)
        \item Drinking water for 85\% of households is regulated this way
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{How I answer these questions}

    \begin{itemize}
        \item Annual panel of drinking water utilities on average 22 km downstream of active coal mines.
        \item Contaminant concentrations for arsenic, nitrate, barium, and selenium, 1997--2005.
        \item Utility self-reporting, regulator visits, and enforcement received.
        \item Instrument upstream coal production.
    \end{itemize}

Why this is appropriate
\begin{itemize}
    \item Separation between contamination source and regulated firm helps identify the effect of contamination.
    \item Coal mining $\rightarrow$ inorganic chemicals (``IOCs'') into surface and ground water.
    \item Geology of coal seams and national energy market determines mining.
\end{itemize}
\end{frame}

\begin{frame}{Utility self-reporting nitrate contamination}

\textbf{Utility manager/staff cognitive burden} \parencite{oleckno1982national}.

Self-reporting requirement for nitrates (one of 83 contaminants):
    \begin{itemize}
        \item \textbf{Baseline}: once a year, send one sample from every intake to a lab
        \item \textbf{Elevated}: nitrate $\geq$ 5 mg/L, half the Maximum Contaminant Level (``MCL''), test quarterly
    \end{itemize}

Miss a test, receive a monitoring and reporting (``MR'') violation and:

\begin{itemize}
    \item Avoid an MCL (need self-report for MCL)
    \item Regulator visit
    \item Or enforcement: Informal (letters, notices, compliance plans); Formal (consent decrees, fines, prosecution)
    \item MR punished less severely than MCL
\end{itemize}
\end{frame}

\begin{frame}{Does coal production raise contamination at drinking water utilities?}
\begin{equation*}
\textit{Concen}_{ct}= \beta\cdot\!\!\sum_{\tau=1985}^{t}\!\!\text{UpstreamCoalProd}_{c\tau} + \rho_c + Watershed_c\cdot\tau_t + \epsilon_{ct}
\end{equation*}
\begin{itemize}\small
    \item $\textit{Concen}_{ct}$: mean annual concentration at utility $c$, year $t$ (SYR2, 1998--2005).
    \item $\beta$: effect of 10 million additional cumulative short tons mined upstream since 1985.
    \item $\rho_c$: utility fixed effects. $Watershed_c\cdot\tau_t$: Watershed-specific year effects, absorbing watershed-wide industrial trends.
    \item SEs clustered at the utility level.
\end{itemize}
\medskip
\begin{itemize}
    \item Continuous treatment, \textbf{no control}: identification comes from relative exposure.
    \item Assumption: utilities with large upstream production changes would have trended like those with small ones. \backuplink{balancetest}{balance test on utility characteristics}
\end{itemize}
\end{frame}

\begin{frame}{IV design: the ARP as a shifter of where coal is mined}
First stage:
\begin{equation*}
\begin{split}
\text{NumMines}_{ct} &= \alpha_0 + \alpha_1\text{NumFacility}_{ct} + \alpha_2\text{UpstreamCoalSulfur\%}_c \cdot \text{Post95}_t \\
&\quad + State_s \cdot Y_t + C_c + \nu_{ct}
\end{split}
\end{equation*}
Second stage:
\begin{equation*}
\text{Outcome}_{vct} = \beta_0 + \beta_1\text{NumFacility}_{ct} + \gamma\widehat{\text{NumMines}}_{ct} + State_s \cdot Y_t + C_c + \epsilon_{ct}
\end{equation*}
\begin{itemize}\small
    \item $\text{Post95}_t = \mathbf{1}\{t \geq 1995\}$. Sulfur is time-invariant, so $C_c$ absorbs its level.
    \item Outcome: 1 if utility $c$ had a violation, visit, or enforcement of type $v$ at any point in year $t$. 
    \item Coefficients are in percentage points.
    \item SEs clustered at the utility level.
\end{itemize}
\end{frame}

% ---------------------------------------------------------------- 23
\begin{frame}{Instrument validity}
\begin{itemize}\small
    \item \textbf{Independence.} The ARP is a national clean-air law set by national energy and emissions politics, not downstream utilities' water quality.
    \item \textbf{Exclusion.} Effect of instrumenting mining downstream of the utility on utility behavior is not significant. \backuplink{kgridmrup}{MR violations}, \backuplink{kgridvisitup}{visits}, \backuplink{kgridenfup}{enforcement}.

    Utility characteristics are not predicted by the instrument. \backuplink{exclfacilities}{full balance table}
    \item \textbf{Monotonicity.} A defier is a high-sulfur watershed that expanded \emph{because} its coal was high sulfur. Scrubber adoption costs more likely lead to lost rather than gained market share; low-cost high-sulfur regions that expand anyway are never-takers. 
    %Monotonicity is pairwise in sulfur, so low- and high-sulfur areas may respond in opposite directions (\cite{angrist1994identification}).
\end{itemize}
\end{frame}

\begin{frame}{Upstream mining reduces utilities' self-reporting}
\presregp{2sls_dwnstrm_minevio_mr_ivsum_binvio_k2}
\takeaway{Base rates of nitrates, arsenic, and IOC are 5.7\%, 3.7\%, and 4.3\%. The 0.56 additional mines a 1-SD lower-sulfur watershed kept after 1995 raise nitrate, arsenic, and IOC MR violations by 2.2, 2.1, and 1.7 pp (state $\times$ year fixed effects). OLS estimates are far smaller --- consistent with negative omitted variable bias, as expected if poorly performing utilities (more MR violations) are protected from mine siting (fewer mines).}
\end{frame}

% ---------------------------------------------------------------- 26
\begin{frame}{Reporting cost, not concealment}
\begin{itemize}\small
    \item Nitrate rules require \textbf{more frequent monitoring} once a reading exceeds 50\% of the MCL --- a discrete jump in cost, with no jump in the penalty for under-reporting and no MCL.
\end{itemize}
\presreg{mr_concentration_lag_ols_k2}
\takeaway{Descriptive, not causal --- read the sign, not the magnitude. Utilities under-report when self-reporting gets expensive, not to conceal exceedances.}
\end{frame}


% ---------------------------------------------------------------- 29
\begin{frame}{The regulator increases inspections}
\presregp{h2_snsv_d12_k2}
\takeaway{Inspection visits rise significantly ($+$2.1 pp; baseline inspection rate 3.4\%). The 0.56 additional mines a 1-SD lower-sulfur watershed kept after 1995 raise inspection visits by 1.2 pp. Regulators do not systematically step up the visit types tied to sanctions. Visits would reveal concealment and be associated with higher penalties.}
\end{frame}

% ---------------------------------------------------------------- 28
\begin{frame}{Enforcement does not counteract under-reporting}
\presregp{h3_inf_formal_d12_k2}
\takeaway{Baseline enforcement rates: informal 18.3\%, formal 2.9\%. One more upstream mine reduces formal enforcement by 3.4 pp with utility and year fixed effects alone (significant; the 0.47 additional mines a 1-SD lower-sulfur watershed kept after 1995 reduce it by 1.6 pp), but the effect loses significance once state $\times$ year fixed effects are added.}
\end{frame}
% ---------------------------------------------------------------- 30
\begin{frame}{Conclusion}
\begin{itemize}\small
    \item Exogenous contamination raises the cost of self-reporting, and self-reporting falls --- \textbf{even when utilities remain below MCL}.
    \item Under-reporting due to cost avoidance, rather than concealment of a contamination violation.
    \item Regulators' inspection visits rise significantly --- substituting for utility self-reporting.
    \item Reduction in enforcement along with increases in inspection suggests that contaminant exceedances are not being concealed and MR violations are not strongly punished.
    \item \textbf{Policy.} Raising self-reporting penalties or reducing self-reporting costs could reduce under-reporting.
    %\item \textbf{Policy.} Tightening MCLs does nothing if nobody tests. By Proposition 3 the binding lever is the price of \emph{not} reporting where source water is impaired, and it is not currently being pulled (\cite{malik1993self}; \cite{harford1987self}).
    \item Open question: whether regulator monitoring is more efficient than self-reporting.
\end{itemize}
