\begin{frame}{Acid rain program}
    \begin{itemize}
        \item Passed in 1990, the first 
        \item Reduce acid rain through reductions in sulfur dioxide and nitrogen oxides below 1980 levels.
        \item Targeted mostly coal burning electric utility plants in eastern and Midwestern states.
        \item Used cap and trade and technology incentives.
        \item Phase I (began in 1995), Affected 263 units
        \item Phase II (began in 2000) Added more units, affected over 2,000 units in all. 
    \end{itemize}
\end{frame}

\section{Triple difference regression general violations}

\begin{frame}{Triple difference regression general violations}
    
\begin{equation}
    Violation_{cpt}=\beta_0 + \beta_3MineHUC12_{p}*Post95_t*HUC12sulfur_p + PWS_p + \epsilon_{cpt}
\end{equation}

\begin{itemize}
    \item Violation is a binary variable for if a violation of a drinking water rule c occurred at PWS p at time t.
    \item MineHUC12 is a binary which is one if the PWS draws its water from a HUC12 where coal mining takes place - control PWS's are those that draw their water from a HUC12 which is upstream of a HUC12 where mining takes place and is not itself downstream from a HUC12 where mining takes place.
    \item Post95 is after 1995, HUC12sulfur is a binary which is one if the HUC12 which the PWS draws it's water from has low sulfur.
\end{itemize}

\end{frame}

\begin{frame}{How do regulators and cws's interact}

\begin{itemize}

    \item $wq_1$ cws output water quality
\end{itemize}

    wq_1 = h(regulator,cws)
\end{equation}

\begin{equation}
    regulator = f(X,cws)
\end{equation}

\begin{equation}
    cws = g(Y, wq_0(Z, mining),regulator)
\end{equation}
