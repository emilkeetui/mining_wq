\begin{frame}{Research question}
    \begin{itemize}
        \item What is the effect of contamination on environmental self-reporting?
        \item Study within the context of drinking water utilities self reporting requirements.
        \item Identify exogenous changes in contamination from coal mining.
        \item Unique to other self-reporting context where reporter endogenously controls the contaminant they must report.
        \item Utilities control water treatment and exert influence over source water, my instrument identifies exogenous changes in source water and therefore in contamination.
    \end{itemize}
\end{frame}

\begin{frame}{IV Model: mining on CWS violations}
First stage:
\begin{equation}\label{1ststage2sls}
\begin{split}
\text{NumMines}_{ct} &= \alpha_0 + \alpha_1\text{NumIntakes}_{ct} + \alpha_2\text{UpstreamCoalSulfur\%}_c \cdot \text{Post95}_t \\
&\quad + Y_t + C_c + \nu_{ct}
\end{split}
\end{equation}

Second stage:
\begin{equation}\label{2ndstage2sls}
\begin{split}
\text{Days in Violation}_{vct} &= \beta_0 + \beta_1\text{NumIntakes}_{ct} + \gamma\widehat{\text{NumMines}}_{ct} \\
&\quad + Y_t + C_c + \epsilon_{ct}
\end{split}
\end{equation}

\begin{itemize}
    \begin{itemize}
        \item $Post95_t=$ 1 if year $>$ 1995 
        \item Standard errors are clustered at the CWS level
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{IV identification assumptions}
    \begin{itemize}
        \item Exclusion: $\text{UpstreamCoalSulfur\%}_c \cdot \text{Post95}_t$ did not affect violations outside of number of upstream mines.
        \item Independence: $\text{UpstreamCoalSulfur\%}_c \cdot \text{Post95}_t$, as good as random conditional on covariates. $\text{Post95}_t$ determined by the national CAAA legislation, and control for time invariant $\text{UpstreamCoalSulfur\%}$ with CWS fixed effects.
        \item Monotonicity/no defiers: A defier is a HUC12 that experienced more mining after 1995 because it had higher sulfur.
    \end{itemize}
\end{frame}

\begin{frame}{DiD Model: mining on CWS contaminant concentrations}
\begin{itemize}
    \item Panel data
    \begin{equation}\label{eq:huc02reg}
    \textit{Concen}_{cht}= \beta\cdot\sum_{\tau=1985}^t\text{UpstreamCoalProd}_{c\tau}  + \rho_c + HUC02_h\cdot\tau_t + \epsilon_{cht}
\end{equation}

    \begin{itemize}
        \item $\textit{Concen}_{cht}=$ mean contaminant concentration at CWS c in HUC02 h in year t.
        \item $\sum_{\tau=1985}^t\text{upstream coal prod}_{c\tau}=$ cumulative tons of coal mined upstream of CWS c in the HUC12 h from 1985 to year t.
        \item $\rho_c=$CWS fixed effects
        \item $HUC02_h\cdot\tau_t=$HUC02 interacted with time fixed effects
        \item Standard errors are clustered at the CWS level
    \end{itemize}
    \item Continuous DiD with no pure control observations. 
    \item Identification: CWSs that experience a large decline in coal production would have trended the same as CWSs with a small decline.
\end{itemize}
\end{frame}

\begin{frame}{Results}
\begin{itemize}
    \item Mining increases mean inorganic chemical contamination (by 40-80\% of mean concentrations).
    \item Below MCL but above CWS contamination preferences.
    \item Mining raises CWS costs to completely remove contaminants.
    \item CWSs downstream of mines face costlier SDWA compliance. 
\end{itemize}
\end{frame}

\begin{frame}{Conclusion}
\begin{itemize}
    \item Mining induces lower CWS SDWA compliance.
    \item $\uparrow$ mining $\rightarrow$   $\uparrow$ inorganic chemical water contamination.
    \item Below MCL but still above CWS preferences.
    \item $\uparrow$ inorganic chemical water contamination $\rightarrow$ $\uparrow$ CWS cost of SDWA compliance.
    \item $\uparrow$ inorganic chemical water contamination $\rightarrow$ $\uparrow$ increasing regulator scrutiny.    
    \item $\uparrow$ mining $\rightarrow$ $\uparrow$ sanitary visits (non-punitive enforcement).
    \item $\uparrow$ mining $\rightarrow$ $\uparrow$ tests.
    \item Testing burden increases, low penalty for missing new testing requirements.
    \item (Lower punishment for violations + higher costs of compliance) $\rightarrow$ reduced CWS SDWA compliance.
\end{itemize}
\end{frame}
