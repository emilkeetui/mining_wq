\section{Introduction}

\begin{frame}{Acid Rain Program (ARP)}
    \begin{itemize}
        \item Reduce acid rain by reducing sulfur dioxide and nitrogen oxides emissions from coal burning electric utility plants.
        \begin{itemize}
            \item Passed in 1990 under the Clean Air Act amendments.
            \item Phase I (began in \textbf{1995}) affected 263 units (most major coal fired power plants).
            \item Phase II (began in 2000) added more units.
        \end{itemize}         
        \item Primary compliance was fuel switching from high- to low-sulfur coal (\cite{schmalensee1998interim}; \cite{schmalensee2013so2}).
        \item Production of high sulfur coal ($\geq 2\%$ sulfur content) fell relative to low sulfur coal.
        \item We use the differential effect of the ARP on coal production to study \begin{itemize}
            \item The impact of coal mining on drinking water quality.
            \item The unintended consequences of the ARP.
        \end{itemize} 
        \item ARP x Coal Sulfur \% as an IV for number of coal mines from \cite{stratford2015coalprod}.
    \end{itemize}
\end{frame}

\begin{frame}{Mining is increasing in the US}
    \begin{itemize} 
        \item A third of global electricity is generated by coal (\cite{IEA2025GER}).
        \item The Trump administration has prioritized maintaining and expanding coal power generation and mining
        \begin{itemize}
            \item Trump signed several pro-coal executive orders: \cite{Trump2025EO14260}, \cite{Trump2025EO14261}, \cite{Trump2025Proc10914}.
            \item And opened millions of acres of federal land for coal mining (\cite{Daly2025CoalMining}).
        \end{itemize}
        \item The environmental impact of coal mining has also been documented (\cite{muller2011environmental}).
        \item US industrial policy seeks to increase critical mineral mining in the US.
        \item \textbf{Community water systems, which provide for 85\% of US households, are the most important public health defenses against water pollution from mining.}
    \end{itemize}
\end{frame}

\begin{frame}{This project}
    \begin{itemize}
        \item \textbf{Research question:} What impact does coal mining have on Community Water Systems' (CWS) behavior?
        \item \textbf{Empirical approach:} 
        \begin{itemize}
            \item Data on CWS Safe Drinking Water Act (SDWA) violations, water intakes, and coal quality and production upstream of intakes from 1985 to 2005.
            \item IV for coal mines upstream of CWS using the differential impact of the Acid Rain Program on coal production based on sulfur percentage.
        \end{itemize}
            \item \textbf{What we find:} An additional coal mine increases SDWA drinking water quality violations.
        \begin{itemize}
            \item Driven by violations from failing to test and submit water quality tests to regulators.
            \item Violations from exceeding maximum contaminant levels (MCLs) are economically and statistically insignificant.
            \item This pattern provides evidence that CWSs increase negligence as water quality deteriorates from upstream industrial activity.
            \item Dangerous for health since MCL rather than MR violations trigger protective avoidance behavior (\cite{zivin2011water}).
        \end{itemize}.
        \item Counterfactual structural modeling of CWS and regulator interactions suggests that to offset the increase in violations from the ARP, enforcement rates need to increase by XX\%. 
    \end{itemize}
\end{frame}

\begin{frame}{The Safe Drinking Water Act}
\begin{itemize}
    \item \textbf{Structure:} EPA sets MCLs, treatment techniques, and monitoring rules;
          primacy states enforce; CWSs test, treat, and report
    \item \textbf{Violations:} monitoring/reporting (MR) if tests not submitted;
          MCL if contaminants exceed regulatory limits
    \item \textbf{Contaminant classes:} inorganic chemicals, VOCs, SOCs,
          radionuclides, bacterials — governed by $\sim$20 rule changes, 1985–2005
    \item \textbf{Primacy states} receive test results, set compliance schedules,
          and issue exemptions/variances — wide discretionary authority
    \item SDWA sets a \textit{floor}; states may impose stricter standards
\end{itemize}
\end{frame}

\begin{frame}{SDWA Compliance and Monitoring}
\begin{itemize}
    \item \textbf{Three violation types:}
    \begin{itemize}
        \item \textit{MCL:} measured concentration exceeds the health-based standard
        \item \textit{MR:} failure to collect or submit required samples on schedule
        \item \textit{TT:} non-compliance with mandated treatment processes
    \end{itemize}
    \item \textbf{Self-reporting design:} no monitoring $\Rightarrow$ no data —
          a system that does not test incurs only an MR violation, not an MCL violation
    \item \textbf{Stable MCLs (1985–2005):} arsenic, nitrates, and VOCs
          had fixed limits throughout the study period
    \item \textbf{Testing schedules vary by contaminant and source type:}
    \begin{itemize}
        \item Total coliform: monthly (reduced to quarterly for small groundwater systems)
        \item IOCs and VOCs: quarterly baseline (1993–95), then annually
              (surface water) or triennially (groundwater)
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}
\textbf{Delegation structure}
\begin{itemize}
  \item States hold primacy: responsible for monitoring, inspection, and enforcement
  \item Federal backstop: if a state does not act within 30 days of a violation, EPA must issue a compliance order or file civil action
\end{itemize}

\medskip
\textbf{Enforcement sequence}
\begin{itemize}
  \item Detection (self-reporting or inspection) $\to$ informal notice $\to$ formal administrative action $\to$ civil/criminal referral
  \item Civil penalties capped at \$5,000/day pre-1996; raised to \$25,000/day
\end{itemize}

\medskip
\textbf{Flexibility mechanisms}
\begin{itemize}
  \item Variances and exemptions allow non-compliance when treatment costs are prohibitive or health risk is low
  \item Reduced monitoring schedules available for contaminants not detected in initial monitoring period
\end{itemize}

\medskip
\textbf{Deterrence evidence}
\begin{itemize}
  \item A single enforcement fine of the Clean Water Act reduces the statewide violation rate by $\approx$two-thirds in the following year, primarily through spillover to uninspected paper pulp firms \citep{shimshack2005regulator}
\end{itemize}
\end{frame}

\begin{frame}{Theoretical model of CWS negligence}

\begin{itemize}
    \item Adapting a static model of a regulator and a CWS from \cite{kang2021understanding} and \cite{mookherjee1994marginal}
    \item CWSs choose a negligence level $a \in [0,\bar{a}]$, from maximum negligence $\bar{a}$ to full compliance 0, regulators do not observe$a$. 
    \item CWSs face a compliance cost, $\theta$, a random variable drawn from $\Theta$, a strictly increasing and continuously differentiable function that depends on mining such that $\theta (m)$. 
    \item CWSs know $\theta$, but regulators only know $\Theta$. 
    \item CWSs receive a benefit $\theta b(a)$.
    \item The opportunity cost to the CWS of choosing a compliance level $a$ relative to maximum negligence $\bar{a}$ is $\theta [b(\bar{a})-b(a)]$.

    \item Given negligence level $a$, observed violations follow a Poisson process and assuming the CWS is risk neutral the expected penalty for a risk neutral CWS is 
\begin{equation}
    e(a)\equiv exp(-a)\sum^\infty_{k=0}\frac{\epsilon(k)}{k!}a^k.
\end{equation}
\end{itemize}
\end{frame}

\begin{frame}{FOCs and comparative static predictions}
\begin{itemize}
\item The expected payoff to the CWS from negligence level $a$ and the FOC is
\begin{equation}\label{eq:cws_benefitcost}
    -\theta(m)[b(\bar{a})-b(a)]-e(a).
\end{equation}
\begin{equation}\label{cws_foc}
    \theta b'[a(\theta)]=e'[a(\theta)]
\end{equation}
whenever $a(\theta)>0$.

\item Proposition 1: CWSs choose higher negligence levels as compliance costs increase, $\partial a/ \partial \theta>0$.

\[\frac{\partial a}{\partial \theta}=-\frac{b'(a)}{\theta b''(a)-e''(a)}>0\]


\item Proposition 2: More upstream mining raises negligence, $\partial a/\partial m >0$.
\[\frac{\partial a}{\partial m}=-\frac{\theta'(m)b'(a)}{\theta(m) b''(a)-e''(a)}>0\]
\end{itemize}
\end{frame}

\begin{frame}{Increased enforcement to offset MR violations}
\begin{itemize}
\item Proposition 3: Higher enforcement rates for violations increase compliance and reduce the number of monitoring and reporting violations. 
\item Assume higher enforcement rates at every $k$ increase the penalty schedule $\epsilon(k)$ by a fixed amount $\lambda>1$, such that $\lambda\epsilon(k)$, then $\partial a/\partial \lambda<0$.

\[\frac{\partial a}{\partial \lambda}=\frac{e'(a)}{\theta b''(a)-\lambda e''(a)}<0\]

\item The increase in negligence and monitoring and reporting violations following an increase in coal mining in lower sulfur areas due to the Acid Rain Program could have been offset by regulator by increasing enforcement of CWS violations.
\end{itemize}
\end{frame}

\begin{frame}{Empirical Structural Estimation: Counterfactual raising probability of enforcement}
\begin{itemize}
\item We estimate a dynamic discrete choice model in which a CWS in year is under state $s\in \{enf,NoEnf\}$ each CWS-year picks comply or MR-violate, where MR carries an implicit per-period cost $k_{MR}$ (the strategic value of concealment) and shifts the next-period probability of being placed under enforcement by $\Delta \pi(s) \equiv Pr(enf' | s, MR) - Pr(enf' | s, comply)$. 
\item Using the Hotz–Miller CCP estimator we recover $k_MR$ from the data. 
\item The counterfactual: holding the structural primitives — including $k_MR$ — fixed, by how much would the regulator have to raise the probability of enforcement following an MR violation to drive aggregate MR violation rates to half their observed level? To zero?
\end{itemize}
\end{frame}

\section{Identification strategy: ARP x Coal Sulfur \% Instrument}

\begin{frame}{Instrumental Variable Identification Strategy}

First Stage:
\begin{equation}
\text{NumMines}_{ct}=\alpha X_{ct}+ \alpha_1UpstreamCoalSulfur\%_c*Post95_t+ Y_t+C_c+S_s+\nu_{ct}    
\end{equation}

Second Stage:
\begin{equation}
\text{Days in Violation}_{vct}=\beta X_{ct}+\gamma\widehat{\text{NumMines}}_{ct}+Y_t+C_c+S_s+\epsilon_{ct}
\end{equation}

\begin{itemize}
    \item Days in Violation$_{vct}$ of rule type v, at CWS c, in year t.
    \item $UpstreamCoalSulfur\%_c$ is the sulfur percentage of the coal in the coal producing HUC12 upstream of a CWS water intake.
    \item Year, $Y_t$, CWS, $C_c$, and state, $S_s$, fixed effects. 
    \item $Post95_t$ is a binary variable indicating if the year $t$ is 1995 or later.
    \item Threats to OLS identification:
    \begin{itemize}
        \item OVB: CWS ability to influence where mines are sited upstream is likely correlated with the resources available to manage violations.
        \item Reverse causality: poorly performing CWSs may be in already polluted areas where it is easier for mines to pass environmental reviews to operate.
        \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Instrument validity}
    \begin{itemize}
        \item  $ARP \times sulfur$ predicts coal demand because the coal market experienced national structural changes; the reverse causal relationship, where CWS characteristics influence coal demand after 1995, is impossible.
        \item Local CWS characteristics regarding water quality did not affect the scope of the ARP, nor how stringent the ARP law was, nor the degree to which low sulfur coal should be part of the abatement technology; thus, $Cov(\text{political power of CWS}, Coal Sulfur \% \times Post95)=0$
    \end{itemize}
\end{frame}

\begin{frame}{Independence of upstream coal sulfur \% $\times$ post95}
    \begin{itemize}        
        \item Post95 is exogenous to water quality; ARP was not passed to address water quality.
        \item $ARP \times Sulfur$ (the sulfur level after 1995) is independent of omitted variables because it has nothing to do with the local power over source protection.
        \item Sulfur is endogenous to water quality; higher sulfur levels are associated with more sulfide minerals in the rock that oxidize and cause AMD.
        \item Assuming CWS characteristics do not change after 1995, $ARP \times Sulfur$ breaks the dependence of $sulfur$ predicting CWS characteristics.
        \item For a given upstream sulfur level $s$, $CoalSulfur_c\times Post95_t=s$, in $t\geq1995$ and $CoalSulfur_c \times Post95_t=0$ in $t<1995$ for all $c$. Given that CWS characteristics do not change over time, the instrument cannot simultaneously predict the same fixed characteristic with two values, implying $f(CoalSulfur_c \times Post95_t, \text{CWS characteristic})=f(CoalSulfur_c \times Post95_t)f(\text{CWS characteristic})$.
    \end{itemize}
\end{frame}

\begin{frame}{Alternative explanation: increased monitoring burden from pollution}
\begin{itemize}
    \item An MR violation could occur if you test below the MCL limit but above a level that requires increased monitoring and reporting. 
    \item Effort required to test and report increases, increasing MR violation.
    \item These systems are also more likely to experience MCL violations because increases in pollution are still driving the increased MR violations.
\end{itemize}
\end{frame}

\begin{frame}{Are MR and MCL violations negatively correlated?}

\begin{itemize}
\item Strategic substitution predicts $Corr(MR, MCL) < 0$ (more MR leads to fewer MCL in the same year).
\item CWS treatment and monitoring incapacity and monitoring burden predict $Corr(MR, MCL) > 0$. 
\item Outcome construction: Aggregate mining MCL and MR across all mining contaminant categories.
\end{itemize}

$$\text{mining MCL}{c,t+k} = \beta_k \text{mining MR}{c,t} + CWS_c + Year_t + State_s + \varepsilon{ct}$$
 
\begin{itemize}

    \item Also estimated as 2SLS, instrumenting $mining MR share days$ with $\text{post95} \times \text{sulfur}$.

\section{Appendix: First stage monotonicity}

\begin{frame}{Monotonicity}
    \begin{itemize}
    \item From \cite{angrist1994identification}, let $\Psi$ be the support of the instrument $Z_i$. Define for each $z \in \Psi$ a random variable $D_i(z)$
    Monotonicity: For all z, w $\in$ $\Psi$, either $D_i(z) \leq D_i(w)$ for all i or $D_i(z) \geq D_i(w)$ for all i. 
    \item Monotonicity is defined on pairwise elements of $\Psi$ and does not specify how z and w are related. This allows for differential changes in number of mines based on sulfur percent. 
    \item Suppose $sulfur= 1$. It is low sulfur, so the number of mines could increase after 1995; then monotonicity requires that for all i $num\_mines_i(0) \leq num\_mines_i(1)$. Now suppose $sulfur = 4$. It is high sulfur, so the number of mines likely decreases after 1995; then monotonicity requires that for all i $num\_mines_i(0) \geq num\_mines_i(4)$. 
    \item Monotonicity can be satisfied even if $num\_mines_i(0) \leq num\_mines_i(1)$ and $num\_mines_i(0) \geq num\_mines_i(4)$.
\end{itemize}
\end{frame}

\section{Robustness check: count of days in violation Poisson regression}

\begin{frame}{Count-data 2SLS: Poisson and Negative Binomial (Wooldridge 2002)}

The control-function approach proceeds in two steps. In the first stage,
the number of upstream coal mines is regressed linearly on the instrument:
%
\begin{equation}
    M_{pt} = \pi \bigl(\mathrm{post}_{95,t} \times \bar{s}_{h(p)}\bigr)
             + \eta_p + \tau_t + \rho_s + v_{pt},
    \label{eq:count-fs}
\end{equation}
%
where $M_{pt}$ is the number of coal mines upstream of PWSID $p$ in year $t$,
$\bar{s}_{h(p)}$ is the mean sulfur content of coal seams in the intake
watershed, $\eta_p$ are PWSID fixed effects, $\tau_t$ are year fixed effects,
$\rho_s$ are state fixed effects, and $v_{pt}$ is the first-stage residual.

In the second stage, violation-days $Y_{pt} \in \mathbb{Z}_{\geq 0}$ are
modeled as a count outcome. Under Poisson QMLE:
%
\begin{equation}
    \mathrm{E}\bigl[Y_{pt} \mid M_{pt},\, \hat{v}_{pt},\, \eta_p, \tau_t,
    \rho_s\bigr]
    = \exp\!\Bigl(\beta\, M_{pt} + \delta\, \hat{v}_{pt}
                 + \eta_p + \tau_t + \rho_s\Bigr),
    \label{eq:poisson-ss}
\end{equation}
%
and under the Negative Binomial:
%
\begin{equation}
    Y_{pt} \mid M_{pt},\, \hat{v}_{pt},\, \eta_p, \tau_t, \rho_s
    \;\sim\; \mathrm{NegBin}\!\left(
        \exp\!\Bigl(\beta\, M_{pt} + \delta\, \hat{v}_{pt}
                   + \eta_p + \tau_t + \rho_s\Bigr),\; \alpha
    \right),
    \label{eq:negbin-ss}
\end{equation}
%
where $\alpha > 0$ is the overdispersion parameter. In both cases $\hat{v}_{pt}$
is the OLS residual from equation~\eqref{eq:count-fs}. The coefficient $\beta$
is a log incidence rate ratio: $\exp(\beta)$ is the multiplicative effect of one
additional upstream mine on expected violation-days. The included CF residual
$\hat{v}_{pt}$ controls for the endogeneity of $M_{pt}$; a significant $\hat\delta$
(i.e.\ $H_0{:}\;\delta = 0$ rejected) indicates that OLS on equation
\eqref{eq:poisson-ss} without the residual would be inconsistent.
Standard errors are clustered at the PWSID level.
\end{frame}
